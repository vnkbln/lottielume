cmake_minimum_required(VERSION 3.20)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/release.json" LOTTIE_RELEASE_JSON)
string(JSON LOTTIE_BASE_VERSION GET "${LOTTIE_RELEASE_JSON}" base_version)
string(JSON LOTTIE_RELEASE_VERSION GET "${LOTTIE_RELEASE_JSON}" version)
string(JSON LOTTIE_FFGL_MAJOR GET "${LOTTIE_RELEASE_JSON}" ffgl_major)
string(JSON LOTTIE_FFGL_MINOR GET "${LOTTIE_RELEASE_JSON}" ffgl_minor)
project(lottie_resolume VERSION ${LOTTIE_BASE_VERSION} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/release.json")
configure_file(src/Version.hpp.in generated/Version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(THORVG_LIBRARY "${CMAKE_SOURCE_DIR}/build-thorvg/src/libthorvg-1.a" CACHE FILEPATH "ThorVG static library")
add_library(thorvg STATIC IMPORTED)
set_target_properties(thorvg PROPERTIES IMPORTED_LOCATION "${THORVG_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/deps/thorvg/inc" INTERFACE_COMPILE_DEFINITIONS TVG_STATIC)
add_library(LottieCore STATIC src/Core.cpp src/Player.cpp)
target_include_directories(LottieCore PUBLIC include PRIVATE build-thorvg deps/thorvg/src/common deps/thorvg/src/renderer deps/thorvg/src/loaders/lottie deps/json/single_include)
target_link_libraries(LottieCore PRIVATE thorvg)
target_compile_definitions(LottieCore PRIVATE NOMINMAX)
set_target_properties(LottieCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(lottie-render src/render.cpp src/png.cpp)
target_link_libraries(lottie-render PRIVATE LottieCore)
include(CTest)
add_executable(core-tests tests/core.cpp)

 add_library(LottieGL STATIC src/Core.cpp)

 target_compile_definitions(LottieGL PRIVATE LOTTIE_OPENGL_RENDERER NOMINMAX)
 target_link_libraries(LottieGL PRIVATE thorvg-gl opengl32)

 target_include_directories(LottiePlayerCPU PRIVATE "${CMAKE_BINARY_DIR}/generated" deps/ffgl/source/lib deps/ffgl/source/lib/glsdk_0_5_2/glload/include)
 target_compile_definitions(LottiePlayerCPU PRIVATE TARGET_OS_WIN)
 target_link_libraries(LottiePlayerCPU PRIVATE LottieCore opengl32 comdlg32)
 set_target_properties(LottiePlayerCPU PROPERTIES OUTPUT_NAME LottiePlayer RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release/cpu" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release/cpu" ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release/cpu" PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release/cpu")
 add_library(LottiePlayer MODULE src/Plugin.cpp deps/ffgl/source/lib/FFGLSDK.cpp)
 target_include_directories(LottiePlayer PRIVATE "${CMAKE_BINARY_DIR}/generated" deps/ffgl/source/lib deps/ffgl/source/lib/glsdk_0_5_2/glload/include)
 target_compile_definitions(LottiePlayer PRIVATE TARGET_OS_WIN LOTTIE_GPU_PLUGIN)
 target_link_libraries(LottiePlayer PRIVATE LottieGL opengl32 comdlg32)
endif()
